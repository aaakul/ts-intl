# Error Handling & Fallback

Configure error callbacks and missing message fallback strategies during initialization:

```ts
import { createI18n, I18nError, I18nErrorCode } from "@aaakul/ts-intl";

export const i18n = createI18n({
  defaultLanguage: "en-US",
  messages: { "en-US": {} },
  onError(error: I18nError) {
    if (error.code === I18nErrorCode.MISSING_MESSAGE) {
      console.warn(
        `[ts-intl] Missing translation key "${error.key}" (language: ${error.lang})`,
      );
    }
  },
  getMessageFallback({ key }) {
    return `[Missing translation: ${key}]`;
  },
});
```
